// Weapon Transformation System
public class WeaponForm : MonoBehaviour
{
    [SerializeField] private WeaponType currentForm;
    [SerializeField] private GameObject[] forms;
    [SerializeField] private float transformDuration = 0.3f;

    private bool isTransforming = false;

    public void SwitchForm(WeaponType newForm)
    {
        if (newForm == currentForm || isTransforming) return;
        StartCoroutine(TransformCoroutine(newForm));
    }

    private IEnumerator TransformCoroutine(WeaponType newForm)
    {
        isTransforming = true;
        
        // Play VFX
        PlayTransformEffect();
        
        yield return new WaitForSeconds(transformDuration);
        
        forms[(int)currentForm].SetActive(false);
        currentForm = newForm;
        forms[(int)currentForm].SetActive(true);
        
        isTransforming = false;
    }
}
// Enemy AI Behavior
public class NightmareAI : MonoBehaviour
{
    [SerializeField] private float detectionRange = 10f;
    [SerializeField] private float attackRange = 2f;
    
    private Transform player;
    private AIState currentState;

    private void Update()
    {
        float distanceToPlayer = Vector3.Distance(transform.position, player.position);
        
        switch (currentState)
        {
            case AIState.Patrol:
                if (distanceToPlayer < detectionRange)
                    currentState = AIState.Chase;
                break;
                
            case AIState.Chase:
                ChasePlayer();
                if (distanceToPlayer < attackRange)
                    currentState = AIState.Attack;
                break;
                
            case AIState.Attack:
                AttackPlayer();
                break;
        }
    }
}
// VFX Controller for Abilities
public class VFXController : MonoBehaviour
{
    [SerializeField] private ParticleSystem[] hitEffects;
    [SerializeField] private ParticleSystem transformEffect;
    
    private ObjectPool effectPool;

    private void Start()
    {
        InitializePool();
    }

    public void PlayHitEffect(Vector3 position, int effectIndex)
    {
        var effect = effectPool.Get();
        effect.transform.position = position;
        effect.Play();
        
        StartCoroutine(ReturnToPoolAfterPlay(effect));
    }

    private IEnumerator ReturnToPoolAfterPlay(ParticleSystem effect)
    {
        yield return new WaitForSeconds(effect.main.duration);
        effectPool.Return(effect);
    }
}
